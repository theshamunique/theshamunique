js = r'''(() => {
  const shells = document.querySelectorAll("[data-marquee]");
  shells.forEach((shell) => {
    const track = shell.querySelector(".marquee-track");
    const firstGroup = track?.querySelector(".marquee-group");
    if (!track || !firstGroup) return;

    // Duplicate the cards once to create a seamless loop.
    const clone = firstGroup.cloneNode(true);
    clone.setAttribute("aria-hidden", "true");
    clone.querySelectorAll("[tabindex]").forEach((el) => el.removeAttribute("tabindex"));
    track.appendChild(clone);

    const allCards = track.querySelectorAll(".skill-card");
    const titleOutput = document.querySelector("[data-detail-title]");
    const copyOutput = document.querySelector("[data-detail-copy]");

    const selectCard = (card) => {
      track.querySelectorAll(".skill-card").forEach((item) => item.classList.remove("is-selected"));
      // If the selected card belongs to the visual duplicate, highlight its matching original.
      const title = card.dataset.title || card.querySelector("h3")?.textContent || "Selected item";
      const detail = card.dataset.detail || card.querySelector("p")?.textContent || "";
      const matchingOriginal = Array.from(firstGroup.querySelectorAll(".skill-card"))
        .find((item) => (item.dataset.title || item.querySelector("h3")?.textContent) === title);
      (matchingOriginal || card).classList.add("is-selected");
      if (titleOutput) titleOutput.textContent = title;
      if (copyOutput) copyOutput.textContent = detail;
    };

    allCards.forEach((card) => {
      card.addEventListener("mouseenter", () => selectCard(card));
      card.addEventListener("focus", () => selectCard(card));
      card.addEventListener("click", () => selectCard(card));
      card.addEventListener("keydown", (event) => {
        if (event.key === "Enter" || event.key === " ") {
          event.preventDefault();
          selectCard(card);
        }
      });
    });

    // Focus within the track pauses the CSS animation; this also works for keyboard users.
    track.addEventListener("focusin", () => { track.style.animationPlayState = "paused"; });
    track.addEventListener("focusout", (event) => {
      if (!track.contains(event.relatedTarget) && !shell.matches(":hover")) {
        track.style.animationPlayState = "";
      }
    });
    shell.addEventListener("mouseenter", () => { track.style.animationPlayState = "paused"; });
    shell.addEventListener("mouseleave", () => {
      if (!track.contains(document.activeElement)) track.style.animationPlayState = "";
    });
  });
})();
'''
